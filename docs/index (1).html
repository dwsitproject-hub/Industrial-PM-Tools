First-time Supabase configuration
Connect to Supabase
Enter your Supabase project credentials below. This only runs once. Find these in your Supabase dashboard → Project Settings → API.
Step 1 — Create Supabase project
Go to supabase.com → New project → fill in name & password → Create.
Step 2 — Run this SQL in Supabase SQL Editor
Copy the SQL below, paste into Supabase → SQL Editor → Run.
-- Run this in Supabase SQL Editor
CREATE TABLE workspaces (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, company text NOT NULL, subtitle text, pw_manager text NOT NULL, pw_admin text NOT NULL, pw_member text NOT NULL, created_at timestamptz DEFAULT now());

CREATE TABLE members (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, workspace_id uuid REFERENCES workspaces(id) ON DELETE CASCADE, name text NOT NULL, color int DEFAULT 0, role text DEFAULT 'member', created_at timestamptz DEFAULT now());

CREATE TABLE site_admins (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, workspace_id uuid REFERENCES workspaces(id) ON DELETE CASCADE, site_name text NOT NULL, password text NOT NULL, color int DEFAULT 3, created_at timestamptz DEFAULT now());

CREATE TABLE tickets (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, workspace_id uuid REFERENCES workspaces(id) ON DELETE CASCADE, ticket_no text, name text NOT NULL, type text, requestor text, priority text DEFAULT 'Normal', status text DEFAULT 'New' CHECK (status IN ('New','In Progress Estimation','In Progress Tender','Done','Hold')), assignee text, deadline date, source text, description text, submitted_by text, site_name text, created_at timestamptz DEFAULT now(), updated_at timestamptz DEFAULT now());

CREATE TABLE notes (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, ticket_id uuid REFERENCES tickets(id) ON DELETE CASCADE, author text, content text NOT NULL, status_at_time text, created_at timestamptz DEFAULT now());

ALTER TABLE workspaces ENABLE ROW LEVEL SECURITY;
ALTER TABLE members ENABLE ROW LEVEL SECURITY;
ALTER TABLE site_admins ENABLE ROW LEVEL SECURITY;
ALTER TABLE tickets ENABLE ROW LEVEL SECURITY;
ALTER TABLE notes ENABLE ROW LEVEL SECURITY;

CREATE POLICY "allow_all" ON workspaces FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON members FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON site_admins FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON tickets FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON notes FOR ALL USING (true) WITH CHECK (true);
